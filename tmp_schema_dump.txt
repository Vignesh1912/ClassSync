CREATE TABLE alembic_version (
	version_num VARCHAR(32) NOT NULL, 
	CONSTRAINT alembic_version_pkc PRIMARY KEY (version_num)
);

CREATE TABLE users (
	id INTEGER NOT NULL, 
	name VARCHAR(100) NOT NULL, 
	email VARCHAR(120) NOT NULL, 
	password_hash VARCHAR(128), 
	role VARCHAR(7) NOT NULL, 
	department VARCHAR(100), 
	college_user_id VARCHAR(50) NOT NULL, 
	is_active BOOLEAN, 
	created_at DATETIME, is_verified BOOLEAN, 
	PRIMARY KEY (id)
);

CREATE TABLE subjects (
	id INTEGER NOT NULL, 
	subject_code VARCHAR(20) NOT NULL, 
	subject_name VARCHAR(150) NOT NULL, 
	credits INTEGER NOT NULL, 
	semester INTEGER NOT NULL, 
	department VARCHAR(100) NOT NULL, 
	PRIMARY KEY (id), 
	UNIQUE (subject_code)
);

CREATE TABLE students (
	id INTEGER NOT NULL, 
	user_id INTEGER NOT NULL, 
	class_section VARCHAR(20), 
	semester INTEGER, 
	roll_number VARCHAR(50), 
	batch VARCHAR(50), 
	PRIMARY KEY (id), 
	FOREIGN KEY(user_id) REFERENCES users (id)
);

CREATE TABLE teachers (
	id INTEGER NOT NULL, 
	user_id INTEGER NOT NULL, 
	designation VARCHAR(100), 
	cabin_no VARCHAR(50), 
	PRIMARY KEY (id), 
	FOREIGN KEY(user_id) REFERENCES users (id)
);

CREATE TABLE timetable (
	id INTEGER NOT NULL, 
	teacher_id INTEGER NOT NULL, 
	subject_id INTEGER NOT NULL, 
	day_of_week VARCHAR(20) NOT NULL, 
	start_time TIME NOT NULL, 
	end_time TIME NOT NULL, 
	room_number VARCHAR(50) NOT NULL, 
	class_section VARCHAR(20) NOT NULL, 
	is_active BOOLEAN, 
	created_at DATETIME, 
	PRIMARY KEY (id), 
	FOREIGN KEY(teacher_id) REFERENCES teachers (id), 
	FOREIGN KEY(subject_id) REFERENCES subjects (id)
);

CREATE TABLE attendance_sessions (
	id INTEGER NOT NULL, 
	teacher_id INTEGER NOT NULL, 
	subject_id INTEGER NOT NULL, 
	session_date DATE NOT NULL, 
	start_time DATETIME NOT NULL, 
	end_time_limit DATETIME NOT NULL, 
	classroom_name VARCHAR(100) NOT NULL, 
	latitude FLOAT NOT NULL, 
	longitude FLOAT NOT NULL, 
	radius_meters FLOAT, 
	status VARCHAR(20), 
	total_present INTEGER, 
	created_at DATETIME, 
	PRIMARY KEY (id), 
	FOREIGN KEY(subject_id) REFERENCES subjects (id), 
	FOREIGN KEY(teacher_id) REFERENCES teachers (id)
);

CREATE TABLE attendance (
	id INTEGER NOT NULL, 
	session_id INTEGER NOT NULL, 
	student_id INTEGER NOT NULL, 
	marked_at DATETIME, 
	student_latitude FLOAT NOT NULL, 
	student_longitude FLOAT NOT NULL, 
	distance_from_class FLOAT NOT NULL, 
	status VARCHAR(20), 
	is_valid BOOLEAN, 
	ip_address VARCHAR(45), 
	PRIMARY KEY (id), 
	FOREIGN KEY(session_id) REFERENCES attendance_sessions (id), 
	FOREIGN KEY(student_id) REFERENCES students (id), 
	CONSTRAINT uix_session_student UNIQUE (session_id, student_id)
);

CREATE TABLE session_activity_log (
	id INTEGER NOT NULL, 
	session_id INTEGER NOT NULL, 
	student_id INTEGER, 
	action VARCHAR(50) NOT NULL, 
	reason VARCHAR(255), 
	latitude FLOAT, 
	longitude FLOAT, 
	ip_address VARCHAR(45), 
	logged_at DATETIME, 
	PRIMARY KEY (id), 
	FOREIGN KEY(session_id) REFERENCES attendance_sessions (id), 
	FOREIGN KEY(student_id) REFERENCES students (id)
);

CREATE TABLE notification_prefs (
	id INTEGER NOT NULL, 
	user_id INTEGER NOT NULL, 
	email_enabled BOOLEAN, 
	remind_3days BOOLEAN, 
	remind_2days BOOLEAN, 
	remind_1day BOOLEAN, 
	remind_deadline BOOLEAN, 
	PRIMARY KEY (id), 
	FOREIGN KEY(user_id) REFERENCES users (id), 
	UNIQUE (user_id)
);

CREATE TABLE assignments (
	id INTEGER NOT NULL, 
	teacher_id INTEGER NOT NULL, 
	subject_id INTEGER NOT NULL, 
	title VARCHAR(255) NOT NULL, 
	description TEXT, 
	deadline DATETIME NOT NULL, 
	classroom_link VARCHAR(500), 
	resource_path VARCHAR(255), 
	created_at DATETIME, 
	PRIMARY KEY (id), 
	FOREIGN KEY(subject_id) REFERENCES subjects (id), 
	FOREIGN KEY(teacher_id) REFERENCES teachers (id)
);

CREATE TABLE submissions (
	id INTEGER NOT NULL, 
	assignment_id INTEGER NOT NULL, 
	student_id INTEGER NOT NULL, 
	self_reported BOOLEAN, 
	submitted_at DATETIME, 
	status VARCHAR(20), 
	PRIMARY KEY (id), 
	FOREIGN KEY(assignment_id) REFERENCES assignments (id), 
	FOREIGN KEY(student_id) REFERENCES students (id), 
	CONSTRAINT uix_assignment_student UNIQUE (assignment_id, student_id)
);

CREATE TABLE announcements (
	id INTEGER NOT NULL, 
	posted_by INTEGER NOT NULL, 
	title VARCHAR(255) NOT NULL, 
	content TEXT NOT NULL, 
	priority VARCHAR(20) NOT NULL, 
	target_audience VARCHAR(20) NOT NULL, 
	target_class VARCHAR(100), 
	target_department VARCHAR(100), 
	created_at DATETIME, 
	PRIMARY KEY (id), 
	FOREIGN KEY(posted_by) REFERENCES users (id)
);

CREATE TABLE notifications (
	id INTEGER NOT NULL, 
	user_id INTEGER NOT NULL, 
	type VARCHAR(50) NOT NULL, 
	title VARCHAR(255) NOT NULL, 
	message TEXT NOT NULL, 
	is_read BOOLEAN, 
	created_at DATETIME, 
	PRIMARY KEY (id), 
	FOREIGN KEY(user_id) REFERENCES users (id)
);

CREATE TABLE announcement_reads (
	id INTEGER NOT NULL, 
	announcement_id INTEGER NOT NULL, 
	user_id INTEGER NOT NULL, 
	read_at DATETIME, 
	PRIMARY KEY (id), 
	FOREIGN KEY(announcement_id) REFERENCES announcements (id), 
	FOREIGN KEY(user_id) REFERENCES users (id), 
	CONSTRAINT uix_ann_user UNIQUE (announcement_id, user_id)
);
